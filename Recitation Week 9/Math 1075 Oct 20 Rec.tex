\section{Intersections and Unions of Sets}
\vspace{1.5em}


1. Given two sets $P=\{a,b,c,d,e,f,g,h,i\}$ and $Q=\{a,e,i,o,u\},$ find

\begin{enumerate}[label=\alph*)]
\item $P\cup Q$
\item $P\cap Q$
\end{enumerate}

\vspace{10em}

\noindent 2. Given sets $A=\{x\mid x<-4\},$ $B=\{x\mid x>2\}$ and $C=\{x\mid 0\leq x\leq 5\},$ find 
\begin{enumerate}[label=\alph*)]
\item $A\cup B$
\item $A\cap B$
\item $A\cup C$
\item $A\cap C$
\item $B\cup C$
\item $B\cap C$
\end{enumerate}
\newpage 
\noindent 3. Find unions and intersections of sets as indicated. 
\begin{enumerate}[label=\alph*)]
\item $(-\infty, 4)\cap [-1,5)$
\item $(-\infty, 4)\cup [-1,5)$
\end{enumerate}
